当前 Database 类关键接口（从 src/storage/database.py 提取）：
```python
class Database:
    def __init__(self, db_path):  # 创建目录 + 初始化表
    def _get_conn(self):          # 返回新 SQLite 连接（每次调用新建）
    def _init_tables(self):       # 创建 hot_topics + articles 表
    def topic_exists(self, title): # 检查热点是否存在
    def insert_topic(self, title, url="", hot_value="", category=""): # 插入热点
    def insert_article(self, topic_id, title, content, style): # 插入文章
    def get_unpublished_articles(self, limit=20): # 获取草稿文章
    def get_recent_articles(self, limit=20): # 获取最近文章
    def mark_published(self, article_id): # 标记文章已发布
```

注意：当前 _get_conn() 每次创建新连接，对 :memory: 数据库不兼容（每次调用会丢失数据）。
db fixture 使用 tmp_path 创建临时文件数据库作为替代方案。

    def test_db_fixture_works(db):
        """验证 db fixture 能创建 Database 实例且表结构正确"""
        conn = db._get_conn()
        tables = conn.execute(
            "SELECT name FROM sqlite_master WHERE type='table'"
        ).fetchall()
        table_names = [t[0] for t in tables]
        conn.close()
        assert 'hot_topics' in table_names
        assert 'articles' in table_names

    def test_db_insert_and_query(db):
        """验证 Database 基本 CRUD 操作"""
        topic_id = db.insert_topic(

        )
        assert topic_id is not None
        assert topic_id > 0
        assert db.topic_exists("测试热点") is True
        assert db.topic_exists("不存在的热点") is False

    def test_db_article_crud(db):
        """验证文章创建和查询"""
        topic_id = db.insert_topic("文章测试热点", "http://example.com", "200", "社会")

        assert article_id is not None
        articles = db.get_recent_articles(limit=5)
        assert len(articles) == 1
        assert articles[0]["title"] == "测试文章标题"
        assert articles[0]["status"] == "draft"
