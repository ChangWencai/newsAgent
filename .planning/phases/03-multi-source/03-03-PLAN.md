       """DailyHotApi HTTP 适配器 - 封装热榜数据获取逻辑"""
       import logging
       import requests
       from typing import Any
       from config.settings import DAILYHOT_BASE_URL

       logger = logging.getLogger(__name__)


       def fetch_dailyhot(platform: str) -> list[dict[str, Any]]:
           url = f"{DAILYHOT_BASE_URL}/{platform}"
           resp = requests.get(url, timeout=15)
           resp.raise_for_status()
           data = resp.json()
           if data.get("code") != 200:
               raise ValueError(f"DailyHotApi 返回异常: platform={platform}, code={data.get('code')}")
           result = data.get("data", [])
           logger.info("DailyHotApi [%s]: 获取 %d 条", platform, len(result))
           return result
       ```
    3. 创建 docker-compose.yml（per CONTEXT.md Specifics 第4点）：

       version: "3.8"
       services:
         dailyhot:
           image: imsyy/dailyhot-api:latest
           container_name: dailyhot-api
           ports:
             - "6688:6688"
           restart: unless-stopped
       ```
